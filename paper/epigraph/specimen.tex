% All three families below are free and distributed by TeX Live.
% Capture each NFSS family after loading its package so they can be compared
% on one sheet without installing any system font.
\usepackage{cinzel}
\edef\CinzelFamily{\rmdefault}
\usepackage{marcellus}
\edef\MarcellusFamily{\rmdefault}
\usepackage{CormorantGaramond}
\edef\CormorantFamily{\rmdefault}

\newcommand{\plainline}{CARPE DIEM QVAM MINIMVM CREDVLA POSTERO}
\newcommand{\dotline}{CARPE\char"00B7DIEM\char"00B7QVAM\char"00B7MINIMVM\char"00B7CREDVLA\char"00B7POSTERO}

% Current successful E306 dedication baseline: 10.5 pt on 16 pt,
% LetterSpace=9 (0.09 em). This is a comparison prior, not a universal rule.
% fontspec tracking is used; there is no manual kerning or synthetic bolding.
\newcommand{\epigraphfont}[2]{%
  \fontfamily{#1}\fontseries{#2}\upshape\selectfont
  \addfontfeatures{LetterSpace=9}%
  \fontsize{10.5}{16}\selectfont
}
\newcommand{\sample}[4]{%
  \par\vspace{7mm}
  {\small\bfseries #1}\par\vspace{2mm}
  \begin{center}
    {\epigraphfont{#2}{#3}#4}
  \end{center}
}
