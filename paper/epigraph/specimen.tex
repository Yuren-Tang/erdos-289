\newcommand{\plainline}{CARPE DIEM QVAM MINIMVM CREDVLA POSTERO}
\newcommand{\dotline}{CARPE\char"00B7DIEM\char"00B7QVAM\char"00B7MINIMVM\char"00B7CREDVLA\char"00B7POSTERO}

% Owner-specified E306 baseline: erdos-306 9ab332be..., whose dedication uses
% EB Garamond Medium at 10.5/16 and LetterSpace=8. This is a controlled starting
% point, not a claim that every family has the same optimal tracking.
\newcommand{\epigraphfont}[2]{%
  \fontfamily{#1}\fontseries{#2}\upshape\selectfont
  \addfontfeatures{LetterSpace=8}%
  \fontsize{10.5}{16}\selectfont
}
\newcommand{\sample}[4]{%
  \par\vspace{7mm}
  {\small\bfseries #1}\par\vspace{2mm}
  \begin{center}
    {\epigraphfont{#2}{#3}#4}
  \end{center}
}
