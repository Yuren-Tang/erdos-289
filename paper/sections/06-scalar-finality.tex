\section{Eventual coverage of component counts}
\label{sec:scalar-finality}

Theorem~\ref{thm:uniform-profile} leaves only two scalar quantities at a prime-power step: the required incoming width $\delta_\rho(Q)$ and, at prime steps, the positive width increment $\mu_\rho(q)$.

Enumerate the prime powers
\[
\{Q=p^e:Q\ge Q_{\mathrm{tr}}\}
\]
in increasing order.  For an integer interval $I=[A,B]$ write $w(I)=B-A$.  At a prime $q$, Theorem~\ref{thm:uniform-profile} gives
\begin{equation}\label{eq:prime-action}
[A,B]\longmapsto
[A+\delta_\rho(q),\,B+\delta_\rho(q)+\mu_\rho(q)],
\end{equation}
so the width increases by $\mu_\rho(q)$.  At a composite prime power $Q=p^e$, $e\ge2$, it gives
\begin{equation}\label{eq:composite-action}
[A,B]\longmapsto[A+\delta_\rho(Q),B],
\end{equation}
so the width decreases by $\delta_\rho(Q)$.  A step is available whenever the incoming width is at least $\delta_\rho(Q)$; then the old and new integer intervals overlap.

Put
\[
\Phi(X)=\frac X{\log X}.
\]
Define
\[
\Delta(X)=\max_{Q\le X}\delta_\rho(Q)
\]
over the displayed prime-power domain, and
\begin{equation}\label{eq:cumulative-debt}
\mathcal L(X)=
\sum_{\substack{p^e\le X\\ e\ge2}}\delta_\rho(p^e).
\end{equation}
By \eqref{eq:d-composite},
\begin{equation}\label{eq:max-demand-subscale}
\Delta(X)=O_\rho((\log X)^2)=o(\Phi(X)).
\end{equation}
The number of prime powers $p^e\le X$ with $e\ge2$ is $O(\sqrt X\log X)$, so
\begin{equation}\label{eq:debt-subscale}
\mathcal L(X)=O_\rho(\sqrt X\,\log^3X)=o(\Phi(X)).
\end{equation}

\subsection{Bounded-lag prime compensation}

Let $\Lambda\ge2$ be the multiplier in Proposition~\ref{prop:prime-supply-main} and put
\begin{equation}\label{eq:sponsor-lag}
\Lambda_s=4\Lambda.
\end{equation}
Every sufficiently large prime-power rank $Q$ admits an earlier prime $q$ such that
\begin{equation}\label{eq:sponsor-bounded-lag}
q<Q\le\Lambda_s q.
\end{equation}
Indeed, let
\[
Y=\left\lfloor\frac{Q}{2\Lambda}\right\rfloor.
\]
For large $Q$, Proposition~\ref{prop:prime-supply-main} gives a prime $Y<q\le\Lambda Y$.  Writing
\[
Q=(2\Lambda)Y+r,
\qquad0\le r<2\Lambda,
\]
shows $q<Q$ and, for $Y\ge1$,
\[
Q<2\Lambda(Y+1)\le4\Lambda Y<4\Lambda q.
\]
After enlarging the onset, $q\ge Q_{\mathrm{tr}}$ as well.  By \eqref{eq:mint-prime},
\begin{equation}\label{eq:sponsor-mint}
\mu_\rho(q)
\gg_\rho\frac q{\log q}
\asymp_{\rho,\Lambda_s}\frac Q{\log Q}
=\Theta_{\rho,\Lambda_s}(\Phi(Q)).
\end{equation}

\subsection{Initial width and continuation}

Let $B_\dagger$ be any prescribed lower cutoff.  Define
\begin{equation}\label{eq:startup-request}
R(B)=\Delta(\Lambda_sB)+\mathcal L(\Lambda_sB).
\end{equation}
By \eqref{eq:max-demand-subscale} and \eqref{eq:debt-subscale},
\begin{equation}\label{eq:startup-subscale-scalar}
R(B)=o(B/\log B).
\end{equation}
Apply Proposition~\ref{prop:wide-start} with $L(B)=R(B)$.  Choose $B$ large enough that $B\ge B_\dagger$, $B\ge Q_{\mathrm{tr}}$, and all preceding asymptotic conclusions hold, and fix the resulting affine initial cover with component-count interval of width $R(B)$ and centre $\gamma_B$.

\begin{lemma}[Continuation]\label{lem:component-count-continuation}
After increasing $B$ if necessary, every prime-power step $Q>B$ is available.  Moreover, immediately before every step $Q>\Lambda_sB$ one has
\begin{equation}\label{eq:sponsored-width}
w(Q)\ge\mu_\rho(q)-\mathcal L(Q)
\end{equation}
for a prime $q$ satisfying \eqref{eq:sponsor-bounded-lag}.
\end{lemma}

\begin{proof}
We induct through the prime powers in increasing order.

First suppose $B<Q\le\Lambda_sB$.  Discarding all positive prime increments, the width immediately before $Q$ is at least
\[
R(B)-\mathcal L(Q)
\ge
\Delta(\Lambda_sB)
\ge
\delta_\rho(Q).
\]
Hence every step in this finite transition is available.

Now let $Q>\Lambda_sB$ and assume all earlier steps have been taken.  Choose $q$ as in \eqref{eq:sponsor-bounded-lag}.  Since $Q\le\Lambda_s q$, we have $q>Q/\Lambda_s>B$, so the prime step $q$ has already occurred after the initial cutoff.  Keeping only its positive increment and discarding every other prime increment gives
\[
w(Q)\ge\mu_\rho(q)-\mathcal L(Q),
\]
which is \eqref{eq:sponsored-width}.  This is a lower bound for the current total width, not an allocation of disjoint credits to later stages; the same earlier prime may therefore witness several stagewise inequalities without being ``spent''.  By \eqref{eq:sponsor-mint} and \eqref{eq:debt-subscale}, the right-hand side is $\gg\Phi(Q)$ for large $Q$, whereas \eqref{eq:max-demand-subscale} gives $\delta_\rho(Q)=o(\Phi(Q))$.  Thus the step $Q$ is available.
\end{proof}

Successive component-count intervals overlap by construction.  For completeness, the union of the first $m$ such integer intervals is itself an integer interval, by induction on $m$ using the successive overlap.  Their upper endpoints are unbounded because the positive increments at prime steps tend to infinity.  The increasing union is therefore an integer interval unbounded above, and hence contains a final ray in $\N$.

\subsection{Vanishing of the affine centre}

The initial affine fibre has centre
\[
\gamma_B\in A/E_B.
\]
Each later prime-power step transports this class along the quotient maps of the endpoint filtration.  By cofinality of $(E_X)$, the image of $\gamma_B$ in $A/E_X$ is zero for all sufficiently large $X$.

Let
\begin{equation}\label{eq:cutoff-base}
\mathcal H_B=\{X\in\N:X\ge B\}.
\end{equation}
For $X\in\mathcal H_B$, let $I_X$ be the component-count interval obtained after all prime-power steps of rank at most $X$ have been processed.  For $k\in\N$ define
\begin{equation}\label{eq:term-k}
\mathcal X_k=
\{X\in\mathcal H_B:
\gamma_B\mapsto0\text{ in }A/E_X,
\quad k\in I_X\}.
\end{equation}

\begin{theorem}[Eventual component-count coverage]\label{thm:scalar-terminality}
For every lower cutoff $B_\dagger$ one can choose $B\ge B_\dagger$ and an affine initial cover at $B$ such that
\[
\exists k_0\ \forall k\ge k_0:
\qquad \mathcal X_k\ne\varnothing.
\]
\end{theorem}

\begin{proof}
Choose $B$ and the affine initial cover by the construction above.  By cofinality, there is $X_0\ge B$ such that the image of $\gamma_B$ in $A/E_X$ is zero for every $X\ge X_0$.  The intervals $I_X$ for $X\ge X_0$ form a tail of the same sequence of successive overlapping intervals, and their upper endpoints are still unbounded.  Hence
\[
\bigcup_{X\ge X_0} I_X
\]
contains $[k_0,\infty)\cap\N$ for some $k_0$.  For any $k\ge k_0$, choose $X\ge X_0$ with $k\in I_X$.  At this $X$ the affine centre has vanished and $k\in I_X$, so $X\in\mathcal X_k$ by \eqref{eq:term-k}.
\end{proof}
