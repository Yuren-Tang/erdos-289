\section{Interval configurations, labelled correspondences, and torsion}
\label{sec:states-filtration}

Let $\mathcal C$ be the set of finite subsets $S\subset\mathbf N_{>0}$ whose connected components in the path graph have cardinality $2$ or $3$ and are pairwise nonadjacent.  Put
\[
W(S)=\sum_{n\in S}\frac1n,
\qquad
\nu(S)=|\pi_0(S)|,
\qquad
\overline W(S)=[W(S)]\in A:=\mathbf Q/\mathbf Z.
\tag{2.1}\label{eq:observations}
\]
For $n\ge1$, let
\[
D_n\subseteq\mathcal C^n
\]
be the joint-compatibility domain: $(S_1,\ldots,S_n)\in D_n$ when the supports are pairwise disjoint and their literal union is again an element of $\mathcal C$.  On $D_n$ the partial product is simply union.  Hence
\[
W\!\left(\bigcup_iS_i\right)=\sum_iW(S_i),
\quad
\nu\!\left(\bigcup_iS_i\right)=\sum_i\nu(S_i),
\quad
\overline W\!\left(\bigcup_iS_i\right)=\sum_i\overline W(S_i).
\tag{2.2}\label{eq:additive-observations}
\]
Equivalently, for binary compatible unions the diagram
\[
\begin{CD}
D_2 @>{\cup}>> \mathcal C\\
@V{(\Omega\pi_1,\Omega\pi_2)}VV @VV{\Omega}V\\
(A\times\mathbf N)^2 @>{+}>> A\times\mathbf N,
\end{CD}
\qquad
\Omega=(\overline W,\nu),
\tag{2.3}\label{eq:observation-square}
\]
commutes.  This is the basic structural square of the proof.

\subsection{Labelled finite correspondences and the relation 2-cell}

We use the standard locally posetal bicategory $\mathbf{Rel}$: its objects are sets, its $1$-morphisms are relations, and a $2$-cell is inclusion of relations.  No higher-categorical structure beyond this convention is needed.

Let $X$ and $Y$ be finite sets.  A \emph{labelled correspondence} from $X$ to $Y$ is a finite subset
\[
R\subseteq X\times\mathcal C\times Y
\]
whose projection to $Y$ is surjective.  Its underlying relation is
\[
U(R)=\{(x,y):\exists S\in\mathcal C,\ (x,S,y)\in R\}\subseteq X\times Y,
\tag{2.4}\label{eq:underlying-relation}
\]
and its reciprocal-mass bound is
\[
\omega(R)=\max\{W(S):(x,S,y)\in R\},
\tag{2.5}\label{eq:response-mass}
\]
with $\omega(\varnothing)=0$.

Suppose $R_1,\ldots,R_n$ form a finite composable string.  Let $P_n$ be the ordinary fibre product of their edge sets over the intermediate target sets.  In general, forgetting labels before composition permits witnesses whose labels are not jointly compatible, so there is only a canonical inclusion
\[
U(R_n\odot\cdots\odot R_1)
\subseteq
U(R_n)\circ\cdots\circ U(R_1).
\tag{2.6}\label{eq:oplax-inclusion}
\]
In $\mathbf{Rel}$ this inclusion is precisely the comparison $2$-cell
\[
\begin{tikzcd}[column sep=huge]
X_0
  \arrow[r,bend left=22,"U(R_n\odot\cdots\odot R_1)"{name=U,below}]
  \arrow[r,bend right=22,swap,"U(R_n)\circ\cdots\circ U(R_1)"{name=D,above}]
& X_n
  \arrow[Rightarrow,from=U,to=D,"\vartheta" description]
\end{tikzcd}
\tag{2.7}\label{eq:oplax-2cell}
\]
(the $2$-cell means inclusion, not a newly defined kind of arrow).

The comparison becomes an identity exactly in the situation used later: if the label map
\[
P_n\longrightarrow\mathcal C^n
\]
factors through $D_n$, then every relationally composable witness is jointly compatible, and
\[
U(R_n\odot\cdots\odot R_1)
=
U(R_n)\circ\cdots\circ U(R_1).
\tag{2.8}\label{eq:rel-strictification}
\]
We shall use \eqref{eq:rel-strictification} only after the factorization through $D_n$ has been proved.  The reciprocal mass is then subadditive:
\[
\omega(R_n\odot\cdots\odot R_1)
\le\sum_{i=1}^n\omega(R_i).
\tag{2.9}\label{eq:mass-subadditive}
\]
Thus the structural calculus consists of ordinary fibre products and images, together with the single oplax comparison \eqref{eq:oplax-inclusion}; joint compatibility turns the latter into strict composition.

\subsection{Finite torsion in \texorpdfstring{$\mathbf Q/\mathbf Z$}{Q/Z}}

For $n\ge1$ put
\[
H_n=\{x\in A:nx=0\}.
\]
For a subgroup $H\le A$, let
\[
q_H:A\longrightarrow A/H
\]
be the quotient map, and for $\alpha\in A/H$ write
\[
\operatorname{Tor}_H(\alpha)=q_H^{-1}(\alpha)
\]
for the corresponding affine torsor fibre.

\begin{lemma}\label{lem:torsion-filtration}
For $m,n\ge1$,
\[
H_n=\langle[1/n]\rangle,
\qquad |H_n|=n,
\]
\[
H_m\le H_n\iff m\mid n,
\qquad
H_m\vee H_n=H_{\operatorname{lcm}(m,n)}.
\]
Consequently the nonzero join-irreducible stages are exactly $H_{p^e}$, with $p$ prime and $e\ge1$.
\end{lemma}

\begin{proof}
If $[q]\in H_n$, then $nq\in\mathbf Z$, so $[q]=[j/n]$ for some $j\in\mathbf Z$; hence $H_n=\langle[1/n]\rangle$ and has order $n$.  The inclusion criterion follows by testing $[1/m]$.  Put $L=\operatorname{lcm}(m,n)$.  Since $\gcd(L/m,L/n)=1$, there exist $a,b\in\mathbf Z$ with
\[
aL/m+bL/n=1.
\]
Dividing by $L$ gives $[1/L]\in H_m+H_n$, whence $H_m\vee H_n=H_L$.  The join-irreducibles of the divisibility lattice are precisely the prime powers.
\end{proof}

Accordingly we index the nontrivial steps by prime powers
\[
Q=p^e.
\]
Let $F_Q$ be the join of all prime-power stages of rank at most $Q$, and $F_{<Q}$ the join of those of smaller rank.  The simple quotient
\[
S_Q=F_Q/F_{<Q}
\tag{2.10}\label{eq:simple-quotient}
\]
has order $p$: the $p$-adic valuation of the lower least common multiple is $e-1$, and adjoining $p^e$ raises it by one.  Thus $S_Q$ is canonically a one-dimensional $\mathbf F_p$-object.

For $X\ge1$ define the endpoint stage
\[
E_X=H_{\operatorname{lcm}(1,\ldots,\lfloor X\rfloor)}.
\tag{2.11}\label{eq:endpoint-stage}
\]
Every class $[a/b]\in A$ belongs to $H_b$, so the filtration $(E_X)$ is cofinal and every fixed class of $A$ becomes zero in $A/E_X$ for all sufficiently large $X$.

Finally, if $Q$ is a prime-power step, then
\begin{equation}\label{eq:endpoint-current-step}
E_{Q-1}=F_{<Q},
\qquad
E_Q=F_Q.
\end{equation}
Indeed a prime-power rank is $<Q$ exactly when it is at most $Q-1$.  Thus the local simple quotient \eqref{eq:simple-quotient} is literally the associated quotient of the global endpoint filtration at $Q$.
