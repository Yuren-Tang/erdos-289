% R10-closed manuscript section: germ projections and root exactification

\section{Filtered response germs and exactification}
\label{sec:germs}

The final exactification argument does not consume the full historical Prefix,
MasterSlab, or BinaryTail objects.  This section records the two regular-image
projections which reduce those stronger providers to the source-filtered germs
actually used by the root.

\subsection{Target restriction of a zero-correction cover}

Let
\[
A=\Q/\Z
\]
and let
\[
H\le G\le A
\]
be finite subgroups.  Write
\[
q_{H,G}:A/H\longrightarrow A/G.
\]
The required residue object for the zero correction from $H$ to $G$ is
canonically
\[
\Req_0(H,G):=\ker q_{H,G}\cong G/H.
\]
No choice of representatives is involved.

\begin{lemma}[Target restriction]\label{lem:target-restriction}
Suppose a normalized physical family $T$ covers the zero correction from $H$
to $G$.  Thus its realizer projection is a regular epimorphism
\[
p_T:\Real_T(H,G;0)\twoheadrightarrow\Req_0(H,G).
\]
Let $S$ be a parameter object and let
\[
r:S\to A/H
\]
be a generalized residue such that
\[
q_{H,G}r=0.
\]
Then there is a canonical pullback response
\[
\begin{tikzcd}
\Real_{T,r} \arrow[r] \arrow[d,two heads]
& \Real_T(H,G;0) \arrow[d,two heads,"p_T"]\\
S \arrow[r,"r"'] & \Req_0(H,G)
\end{tikzcd}
\]
whose left vertical map is regular epi.  Binaryity, component profile,
resource upper bounds, and compatibility with a fixed obstacle are inherited
by restriction.
\end{lemma}

\begin{proof}
The condition $q_{H,G}r=0$ is exactly the universal property which factors
$r$ through $\ker q_{H,G}$.  The displayed square is therefore defined
canonically.  Regular epimorphisms are stable under pullback.  The remaining
properties are predicates or observables inherited by the restricted physical
family.
\end{proof}

For $r\in A/H$, let
\[
H[r]=q_H^{-1}(\langle r\rangle)
\]
be the least finite extension of $H$ killing $r$.  If $G\ge H[r]$, then
$r\in\ker q_{H,G}$ and Lemma~\ref{lem:target-restriction} applies.

\subsection{The late-source targeted Tail germ}

Fix a raw reciprocal provider $R$.  Its eligible source object carries a
proper filter
\[
\mathcal F_R
\]
and every eligible source $\beta$ determines a finite subgroup
\[
H_\beta\le A.
\]

For a row-transparent finite obstacle $O$, a residue
$r\in A/H_\beta$, a tolerance $\varepsilon>0$, and $h\in\N$, let
\[
\LTar_R(\varepsilon,c,\beta,O,r,h)
\]
denote the normalized solution object of binary physical responses with
\[
(b_2,b_3)=(h,0),
\qquad
W_+\le\varepsilon,
\qquad
q_{H_\beta}([W])=r,
\]
compatible with $O$.

\begin{theorem}[Late-source targeted residue saturation]
\label{thm:ltar-closed}
For every $R$, every $c>0$, and every $\varepsilon>0$,
\[
\forall^\infty\beta\,[\mathcal F_R],
\quad
\forall O,
\quad
\forall r\in A/H_\beta,
\quad
\forall^\infty h,
\]
the projection
\[
\LTar_R(\varepsilon,c,\beta,O,r,h)\longrightarrow1
\]
is regular epi.
\end{theorem}

\begin{proof}
Use the stronger historical BinaryTail theorem at the same tolerance and
source.  Once a good source $\beta$ and a residue $r\in A/H_\beta$ have been
fixed, form the canonical least absorber
\[
K_r=H_\beta[r].
\]
Instantiate BinaryTail at the compact request $K_r$.  Its eventual endpoint
$G\ge K_r$ supplies a full zero-correction cover from $H_\beta$ to $G$.
Since $G$ kills $r$, Lemma~\ref{lem:target-restriction} restricts this cover to
the required residue $r$.

No quantifier is interchanged: the good-source locus of BinaryTail is already
uniform in the obstacle and compact request; after fixing $\beta$, the residue
$r$ determines $K_r$, and the eventual grade threshold is allowed to depend
on $r$.  The endpoint, compact request, least absorber and the unused parts of
the zero-correction fibre disappear after this pullback.
\end{proof}

\begin{remark}[Fixed-source smallness is impossible]
The source eventuality in Theorem~\ref{thm:ltar-closed} is essential.  If
\[
H_n=n^{-1}\Z/\Z,
\]
then a positive value $0<W<1$ with zero residue modulo $H_n$ has
$W\ge1/n$.  Hence a fixed source cannot support positive zero-residue states
of arbitrarily small mass.  The tolerance quantifier must precede the eventual
source quantifier.
\end{remark}

\subsection{Projection of MasterSlab to a filtered affine ray}

Let
\[
(B,\mathcal F_B)
\]
be a Prefix base and let
\[
\beta:B\to S_R
\]
be its source map.  The Prefix theorem supplies
\[
\operatorname{Tendsto}(\beta,\mathcal F_B,\mathcal F_R).
\tag{\ref{sec:germs}.1}
\]

Over a generalized good-scale point $(c,\eta)$, the stronger MasterSlab
theorem gives, for all sufficiently late $b$ and every
\[
L\ge d_0(b),
\qquad d_0(b)=\delta_b^{\sup},
\]
a normalized finite family $S_{b,L}$ for which
\[
(\operatorname{required},\delta_3):
S_{b,L}	woheadrightarrow
\operatorname{Required}(\phi_{b,L})
\times[d_0(b),L]
\tag{\ref{sec:germs}.2}
\]
is regular epi.  All branches have one common total grade, reciprocal value
in $(0,2-\eta)$, and the whole family is row-transparent with parameter $c$ at
the source $\beta(b)$.

\begin{proposition}[Affine-ray fibre projection]
\label{prop:affine-ray-projection}
For every sufficiently late $b$ and every $d\ge d_0(b)$ there is a finite
normalized physical family
\[
F_{b,d}	woheadrightarrow q_{H_b}^{-1}(a_{b,d})
\]
covering one affine coset in $A$, with constant component profile
\[
(m_{b,d},d)
\]
for a derived $m_{b,d}\in\N$, reciprocal value in $(0,2-\eta)$, and the same
row-transparency lower bound $c$.
\end{proposition}

\begin{proof}
In the MasterSlab theorem substitute $L=d$.  Pull the regular epimorphism
\eqref{sec:germs}.2 back along the terminal defect fibre
\[
\operatorname{Required}(\phi_{b,d})\times\{d\}
\hookrightarrow
\operatorname{Required}(\phi_{b,d})\times[d_0(b),d].
\]
This gives
\[
S^{\mathrm{fib}}_{b,d}	woheadrightarrow
\operatorname{Required}(\phi_{b,d}).
\]
Because the correction starts at the zero compact stage and ends at $H_b$,
its required object is canonically the affine coset
\[
\operatorname{Required}(\phi_{b,d})
=q_{H_b}^{-1}(a_{b,d}).
\]
The fibre has exact ternary count $d$ and inherits the common total grade
$K_{b,d}$, hence its component profile is
\[
(b_2,b_3)=(K_{b,d}-d,d).
\]
The value margin is branchwise and therefore restricts to the fibre.
Transparency is hereditary in the required direction: restricting an obstacle
can only enlarge the surviving compatible raw row.  All other defect fibres,
selector data, neutral coordinates, and cross-defect commonality are forgotten
after this pullback by regular-image factorization.
\end{proof}

The family of Proposition~\ref{prop:affine-ray-projection}, together with
\eqref{sec:germs}.1, is the \emph{filtered affine ray}.  The root uses no
common affine correction or common affine grade across different defects.

\subsection{Meeting the two source germs}

The generalized margin coordinate $\eta$ supplies the canonical Tail tolerance
\[
\varepsilon=\eta/2.
\]
For fixed $(c,\eta)$, Theorem~\ref{thm:ltar-closed} gives a
$\mathcal F_R$-large set of good Tail sources.  Its inverse image under
$\beta$ is $\mathcal F_B$-large by Tendsto.  Intersecting with the affine-ray
large locus gives another $\mathcal F_B$-large subobject
\[
U_{\mathrm{meet}}
=
U_{\mathrm{aff}}
\cap
\beta^{-1}(U_{\mathrm{tail}}).
\tag{\ref{sec:germs}.3}
\]
Properness makes this intersection nonempty.  We work over a generalized
point $b$ of this covering locus; no preferred base is selected.

Put
\[
d_*=d_0(b).
\]
The source is now held fixed.  This is essential for the present provider:
the Prefix defect floor rises as the source moves later, while the available
Tail resource improves.

\begin{proposition}[Source--defect tradeoff]
\label{prop:source-defect-tradeoff}
For the current Prefix provider,
\[
d_0(b)=\delta_b^{\sup}\longrightarrow\infty
\]
along the moving source filter.  Consequently the source may not be
existentially reselected with the final grade after a fixed defect $d$ has
been chosen.
\end{proposition}

\begin{proof}
For a finite full product family $F$, write
\[
\Delta(F)=\max_{x\in F}\delta_3(x).
\]
Additivity of $\delta_3$ and the presence of every product branch give
\[
\Delta(F\star G)=\Delta(F)+\Delta(G).
\]
Every nontrivial universal bridge contains a Boolean endpoint switch with a
length-three alternative, so its defect supremum is at least one.  The moving
Prefix crosses infinitely many such bridge stages.  Its defect supremum is
therefore the seed supremum plus an unbounded sum of positive bridge costs.
The MasterSlab affine ray begins at this supremum.  Moving the source later can
thus eventually push a fixed defect below the available affine ray.
\end{proof}

\subsection{Coset cancellation}

Fix $d\ge d_*$ and an affine fibre
\[
F_{b,d}	woheadrightarrow q_{H_b}^{-1}(a_d)
\]
with component profile $(m_d,d)$.  Its physical footprint is a finite
row-transparent obstacle.

Apply Theorem~\ref{thm:ltar-closed} at the same source $H_b$, the opposite
residue
\[
-a_d\in A/H_b,
\]
and tolerance $\eta/2$.  For all sufficiently large $h$ obtain a compatible
binary Tail family $T_h$ of profile $(h,0)$ with mass at most $\eta/2$ and
residue $-a_d$.

Form the literal matching pullback over the affine residue:
\[
\begin{tikzcd}
M_{d,h} \arrow[r] \arrow[d,two heads]
& F_{b,d} \arrow[d,two heads,"{q_{H_b}[W]}"]\\
T_h \arrow[r,"{-[W]}"']
& \{a_d\}\subseteq A/H_b.
\end{tikzcd}
\tag{\ref{sec:germs}.4}
\]
Every point of $M_{d,h}$ is an actual compatible pair of physical realizers;
physical union therefore gives a state with
\[
[W(f\sqcup t)]
=[W(f)]+[W(t)]
=a_d-a_d=0
\quad\text{in }A=\Q/\Z.
\]
Moreover
\[
0<W(f)<2-\eta,
\qquad
0\le W(t)\le\eta/2,
\]
so
\[
0<W(f\sqcup t)<2.
\]
Since its reciprocal value is an integer, it is exactly $1$.

Component profiles add:
\[
(m_d,d)+(h,0)=(m_d+h,d).
\]
Thus the realizable binary counts at defect $d$ contain a translate of a final
subset of $\N$, and are themselves final.

\begin{theorem}[Fibrewise profile saturation]
There exists $d_*\in\N$ such that for every $d\ge d_*$ all sufficiently large
$b_2$ occur in the exact component-profile spectrum
\[
\Sigma^{23}_1.
\]
\end{theorem}

\begin{proof}
Choose the generalized common source above once and put $d_*=d_0(b)$.  For each
fixed $d\ge d_*$, the preceding coset-cancellation argument shows that the
realizable binary counts contain $m_d+J_d$, where $J_d$ is final.  Finite
translation preserves finality.
\end{proof}

For Erdős Problem 289 itself no finite meet is required: fix one
$d\ge d_*$; then adding this fixed $d$ to the final binary-count set gives a
final set of total component counts.  Finite intersections over
$d\in[d_*,D]$ give the stronger finite-defect rectangle as a corollary.
